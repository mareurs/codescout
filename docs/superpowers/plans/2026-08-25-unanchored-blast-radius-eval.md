# Un-anchored Blast-Radius Eval Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an eval that measures whether an agent, given a narrow bug whose fix
edits a shared function, generates the sub-goal *"who else depends on this?"* on its
own — and whether codescout's tools change how completely it answers.

**Architecture:** A new scenario directory `scenarios/blast-radius/` in
`prompt-engineering`, generated by extending the existing `gen_fixture.py`. A
one-character defect in `duty_multiplier` is planted; six dependents using six
different reference forms are planted around it; a golden-output harness records every
entry point's value before and after the run; `check_blast.py` scores four metrics
(L0–L3) from the trace and the behaviour, never from an authored answer key.

**Tech Stack:** Python 3.12+, `pytest`, the `prompt-tdd` harness
(`scripts/run_arms.py`, `scenarios/surface-budget/surface_lib.py`), headless
`claude -p` via the Claude Code adapter.

**Spec:** [`docs/superpowers/specs/2026-08-25-unanchored-blast-radius-eval-design.md`](../specs/2026-08-25-unanchored-blast-radius-eval-design.md)
in the `codescout` repo. Read it before Task 1; the plan argues from it.

```python
def test_all_six_dependents_exist_with_their_declared_forms(tree):
    """Six DIFFERENT reference forms. Six copies of one form would make a single
    lexical pattern the complete answer -- an oracle, which is what killed band C's
    first design (`if all four named TAX_RATE, one grep selects 8 of 12 sites`)."""
    cases = [
        ("src/intl/customs.py", "LEVY_MULTIPLIER = duty_multiplier()"),
        ("src/intl/checkout.py", "duty_multiplier()"),
        ("src/intl/manifest.py", "import duty_multiplier as _dm"),
        ("src/orders/crossborder.py", "from src.intl import duty_multiplier"),
        ("src/pricing/registry.py", 'RATE_FNS = {"duty": "duty_multiplier"}'),
        ("src/exports/customs_feed.py", '_resolve("duty")'),
    ]
    for path, needle in cases:
        assert needle in (tree / path).read_text(), f"{path} lost its reference form"


def test_the_config_dispatch_row_is_invisible_to_a_python_only_search(tree):
    """Row 6 is findable ONLY by an agent that searches beyond *.py. A grep whose
    default scope excluded the answer once returned `0 matches` on four probes,
    including one naming the file's exact path as its glob."""
    feed = (tree / "src/exports/customs_feed.py").read_text()
    assert "duty_multiplier" not in feed, "row 6 must not name the function in Python"
    assert "duty_multiplier" in (tree / "pricing.toml").read_text()


def test_every_dependent_changes_when_the_defect_is_fixed(tree, tmp_path):
    """The whole of L3 rests on this. If a 'dependent' does not move when the bug is
    fixed, it is not a dependent and scoring it as one is a fabricated number."""
    import shutil
    fixed = tmp_path / "fixed"
    shutil.copytree(tree, fixed)
    p = fixed / "src/intl/duties.py"
    p.write_text(p.read_text().replace("_rate_bp() // BASIS", "_rate_bp() / BASIS"))

    def golden(root):
        out = subprocess.run([sys.executable, str(HERE / "golden.py")],
                             cwd=root, capture_output=True, text=True, check=True)
        return __import__("json").loads(out.stdout)

    before, after = golden(tree), golden(fixed)
    for name in ("international_total", "apply_levy", "describe_duty",
                 "manifest_line", "landed_cost", "apply_named", "feed_row"):
        assert before[name] != after[name], f"{name} did not move when the bug was fixed"
    for name in ("surcharge_label", "rate_basis_points_label"):
        assert before[name] == after[name], f"{name} moved and must not have"
```

```python
    # Balance is 2 / 2 / 2 -- two forms both toolchains find, two only LSP finds, two
    # only a lexical sweep finds. Neither toolchain reaches 6/6 alone, and the set does
    # not favour either one. Rows 5 and 6 are the ones codescout LOSES, by two
    # different mechanisms so one slip does not decide that case.
    #
    # Rows 1 (customs.LEVY_MULTIPLIER) and 2 (checkout.describe_duty) already exist
    # above. These are rows 3-6.

    # Row 3 -- aliased import. LSP resolves the alias; a grep for `duty_multiplier`
    # hits the import line and MISSES the call site, which reads `_dm()`.
    _write(root, "src/intl/manifest.py", _with_padding(rng, "intl",
           "from src.intl.duties import duty_multiplier as _dm\n\n\n"
           "def manifest_line(amount: float) -> str:\n"
           "    \"\"\"Render a customs manifest line for a declared amount.\"\"\"\n"
           "    return f\"declared={amount * _dm():.2f}\"\n", exports))

    # Row 4 -- package re-export. The caller never names duties.py at all.
    _write(root, "src/intl/__init__.py",
           "from src.intl.duties import duty_multiplier\n")
    _write(root, "src/orders/crossborder.py", _with_padding(rng, "orders",
           "from src.intl import duty_multiplier\n\n\n"

           "    \"\"\"Total cost of a cross-border order once duty is applied.\"\"\"\n"
           "    return subtotal * duty_multiplier()\n", exports))

    # Row 5 -- string in a dict, resolved by getattr. No symbol reference exists, so
    # LSP references cannot see it; a lexical sweep finds the string.
    _write(root, "src/pricing/registry.py", _with_padding(rng, "pricing",
           "import src.intl.duties as _duties\n\n\n"
           "RATE_FNS = {\"duty\": \"duty_multiplier\", \"surcharge\": \"surcharge_pct\"}\n\n\n"

           "    mod, fn = cfg[\"rates\"][key].split(\":\")\n"
```

```python
FORM_PREDICATES = {
    "aliased import":       lambda s: " as _" in s and "import" in s,
    "package re-export":    lambda s: "from src.intl import " in s,
    "getattr on a dict":    lambda s: "getattr(" in s,
    "toml dispatch":        lambda s: "tomllib" in s,
    "import-time call":     lambda s: bool(__import__("re").search(r"(?m)^[A-Z_]+ = \w+\(\)$", s)),
}


@pytest.mark.parametrize("form", sorted(FORM_PREDICATES))
def test_no_reference_form_is_itself_the_answer(tree, form):
    """Each form must appear in at least two files that are NOT dependents.

    Otherwise the FORM is the oracle: one grep for `getattr(` would enumerate the
    dependent set without any search of the call graph. This is the generalisation of
    band C's rule -- `exactly ONE band-C file names TAX_RATE. If all four did,
    'imports src/config/rates.py' would select 8 of 12 truth sites`."""
    dependent_paths = {p for p, _ in gen_fixture.DEPENDENTS}
    pred = FORM_PREDICATES[form]
    decoys = [p for p in tree.rglob("*.py")
              if str(p.relative_to(tree)) not in dependent_paths and pred(p.read_text())]
    assert len(decoys) >= 2, (
        f"form {form!r} appears in only {len(decoys)} non-dependent file(s); "
        "the form is acting as an oracle")
```

```python
"""The harness is the ground truth for L3, so it has to fail loudly rather than
partially. A harness that silently skips an entry point reports 'nothing changed'
for a run that broke it -- the silent-zero shape (`count: 0` against 233 files)."""
import json
import subprocess
import sys
from pathlib import Path

import pytest

import gen_fixture

HERE = Path(__file__).parent


@pytest.fixture(scope="module")
def tree(tmp_path_factory):
    root = tmp_path_factory.mktemp("golden")
    gen_fixture.build(root)
    return root


def _run(root):
    out = subprocess.run([sys.executable, str(HERE / "golden.py")],
                         cwd=root, capture_output=True, text=True)
    return out


def test_it_reports_every_entry_point(tree):
    got = json.loads(_run(tree).stdout)
    expected = {"international_total", "apply_levy", "describe_duty", "manifest_line",
                "landed_cost", "apply_named", "feed_row",
                "surcharge_label", "rate_basis_points_label",
                "compute_total", "grand_total"}
    assert expected <= set(got), f"missing: {expected - set(got)}"


def test_a_broken_entry_point_is_an_error_not_a_missing_key(tree, tmp_path):
    """If an import raises, the harness must exit non-zero. Dropping the key would
    make a crashed entry point indistinguishable from an unchanged one."""
    import shutil
    broken = tmp_path / "broken"
    shutil.copytree(tree, broken)
    (broken / "src/intl/manifest.py").write_text("raise RuntimeError('boom')\n")
    out = _run(broken)
    assert out.returncode != 0
    assert "manifest" in (out.stderr + out.stdout)
```

```python
#!/usr/bin/env python3
"""Score a blast-radius run: L0 fix, L1 asked, L2 enumerated, L3 silent changes."""
import json
import re
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).parents[1] / "surface-budget"))
sys.path.insert(0, str(Path(__file__).parent))

import surface_lib                                        # noqa: E402
from gen_fixture import DEPENDENTS                        # noqa: E402

SHARED = "duty_multiplier"
DEPENDENT_PATHS = [p for p, _ in DEPENDENTS]

# PRE-REGISTERED DETECTORS, both families. Written out rather than inferred, because a
# detector that recognises one family's idiom and not the other's makes the primary
# metric arm-asymmetric by construction -- and it would do so silently, reporting a
# plausible number. Asserted symmetric by gate 3.
_PATH_KEYS = ("file_path", "path", "notebook_path")
_TEXT_KEYS = ("command", "pattern", "symbol", "query", "name", "old_string")


def _paths(call: dict) -> list[str]:
    args = call.get("args") or {}
    out = [str(args[k]) for k in _PATH_KEYS if k in args]
    out += [str(args[k]) for k in _TEXT_KEYS if k in args]
    return out


def l1_asked(trace: list) -> bool:
    """Did ANY call target the shared function or one of its dependents?

    Deliberately generous: a Bash grep counts, a codescout references call counts,
    opening a dependent file counts. The point is to detect that the sub-goal was
    generated at all, not to reward a particular route.
    """
    for call in trace or []:
        blob = " ".join(_paths(call))
        if SHARED in blob or any(d in blob for d in DEPENDENT_PATHS):
            return True
    return False


def l2_enumerated(trace: list) -> int:
    """How many of the six dependents the run actually REACHED, from the trace.

    Not from the answer text. An answer can name a file the run never opened, which is
    the confident-wrong-answer shape this eval exists to catch; and scoring the answer
    would reintroduce the list-shaped metric that saturated the sibling eval.
    """
    seen = set()
    for call in trace or []:
        blob = " ".join(_paths(call))
        for dep in DEPENDENT_PATHS:
            if dep in blob:
                seen.add(dep)
    return len(seen)


def l3(before: dict, after: dict, answer: str) -> dict:
    """Outputs that moved, and how many of them the answer never mentioned."""
    changed = [k for k in before if before.get(k) != after.get(k)]
    disclosed = [k for k in changed if k in answer]
    return {"changed": len(changed), "disclosed": len(disclosed),
            "silent_changes": len(changed) - len(disclosed),
            "changed_names": sorted(changed)}


def reconcile(enumerated: int, disclosed: int) -> dict:
    """Naming more than you looked at is a finding, not a scoring error."""
    return {"disclosed_beyond_enumerated": max(0, disclosed - enumerated)}


def l0_fix_correct(before: dict, after: dict) -> bool:
    """The reported symptom is gone: international_total no longer equals the amount."""
    return before.get("international_total") != after.get("international_total")
```
